จากเว็บที่ตอบได้ สู่เว็บที่จำสถานะได้

ร้านค้าออนไลน์ต้องจำว่าตะกร้าเป็นของใคร แม้ผู้ใช้เปิดหน้าใหม่ และต้องเก็บรายการสินค้าไว้แม้ server หยุดทำงาน บทนี้แยกปัญหาออกเป็น configuration, สถานะชั่วคราว และข้อมูลถาวร แล้วเลือก .env, cookie/session และฐานข้อมูลให้ถูกหน้าที่

ก่อนอ่านควรรู้ route, middleware และ req/res จาก L9–L10 เมื่อจบบทควรอธิบายได้ว่าข้อมูลอยู่ฝั่งไหน คาดการณ์ตัวนับ session และไล่เส้นทาง Express → SQL → JSON ได้

Express ช่วย routing และ API; template engine เช่น EJS/Pug แทนค่าตัวแปรเป็น HTML ก่อนส่งให้ browser ส่วน API อาจส่ง JSON ให้ client ไปสร้างหน้าตาเอง

L11 · หน้า PDF 9–14

ตั้งค่าโครงการโดยไม่ผูกกับเครื่อง

Environment variable คือคู่ชื่อและค่าที่แยก configuration ออกจาก source code เช่น PORT และ MYSQL_HOST โค้ดชุดเดียวจึงทำงานกับเครื่องพัฒนาและเครื่องอื่นได้โดยเปลี่ยนค่า ไม่ต้องแก้ route

npm init -y
npm install express dotenv cookie-parser express-session mysql2
npm install --save-dev nodemon

npm install ติดตั้ง dependencies; npm uninstall ชื่อแพ็กเกจ เอาออก ส่วน npm start เรียก scripts.start เช่น nodemon app.cjs ใน package.json; nodemon มีหน้าที่ restart ระหว่างพัฒนา

// app.cjs (CommonJS)
require('dotenv').config(); // โหลดก่อนอ่าน process.env
const express = require('express');
const app = express();
app.get('/', (req, res) => res.send('Ready'));
const port = Number(process.env.PORT || 3000);
app.listen(port, () => console.log(`Port ${port}`));
# .env: ข้อมูลตัวอย่าง ไม่ใช่ credentials จริง
PORT=3000
MYSQL_HOST=localhost
MYSQL_USERNAME=webapp
MYSQL_PASSWORD=replace-locally
MYSQL_DATABASE=tinycollege

ค่าใน process.env เป็นข้อความ ต้องแปลงเมื่อจำเป็น .env ไม่ใช่การเข้ารหัส: เก็บนอก Git และใช้ .env.example ที่ไม่มี secret แทน สไลด์หน้า 10 พิมพ์ npm init <package> ในขั้นติดตั้ง; คำสั่งที่ถูกคือ npm install <package>

L11 · หน้า PDF 15–29

Cookie เชื่อม request ได้อย่างไร

HTTP เป็น stateless: request ใหม่ไม่ได้มีความทรงจำของ request ก่อนโดยอัตโนมัติ Server จึงส่ง Set-Cookie ใน response ให้ browser เก็บ แล้ว browser ส่ง Cookie กลับตามขอบเขต domain/path และเงื่อนไขของ cookie เพื่อให้แอปเชื่อมคำขอได้

const cookieParser = require('cookie-parser');
app.use(cookieParser()); // ก่อน route ที่อ่าน req.cookies
app.get('/set-cookie', (req, res) => {
  res.cookie('theme', 'dark', { maxAge: 360000, sameSite: 'lax' });
  res.send('Cookie sent');
});
app.get('/show-cookie', (req, res) => res.json(req.cookies));
app.get('/clear-cookie', (req, res) => {
  res.clearCookie('theme');
  res.send('Cookie cleared');
});

res.cookie() สร้าง header ส่วน cookie-parser อ่าน header ที่มากับ request เป็น req.cookies ไม่ได้เป็นสิ่งจำเป็นสำหรับการส่ง Set-Cookie เอง ตรวจใน DevTools → Application → Cookies แล้วเปิด /show-cookie ใน request ถัดไปจึงจะเห็นค่าที่ส่งกลับ

ตัวเลือกความหมาย
maxAgeอายุเป็นมิลลิวินาที เช่น 360000 = 6 นาที
expiresวันหมดอายุ ใช้ Date เช่น new Date(Date.now() + 360000)
httpOnlyไม่ให้ JavaScript ในหน้าอ่าน cookie ผ่าน document.cookie
secureส่งผ่าน HTTPS เป็นหลัก เหมาะกับระบบจริง
sameSiteกำหนดเงื่อนไขส่ง cookie ข้าม site ช่วยลดบางกรณีของ CSRF

อธิบายเสริม: สไลด์หน้า 26 ใช้ expire; Express ใช้ expires และรับ Date การ clear ต้องตรงชื่อและขอบเขตเดิม เช่น path/domain การมี cookie ชื่อ tokenID อย่างเดียวตาม demo ไม่ได้พิสูจน์ว่า login ถูกต้อง ผู้ใช้แก้ข้อมูลฝั่ง client ได้ ต้องตรวจหลักฐานฝั่ง server

L11 · หน้า PDF 30–34

Session: browser ถือบัตร ส่วนข้อมูลอยู่ server

Session เก็บสถานะบน server เช่น numviews โดย browser ถือเพียง session ID ที่ใช้ค้นข้อมูล ไม่ได้ถือ object session ทั้งก้อน express-session จัดการ ID และการเข้าถึงผ่าน req.session; secret ใช้ลงลายเซ็น ID ไม่ใช่เข้ารหัสข้อมูลทุกอย่าง

const session = require('express-session');
app.use(session({
  secret: process.env.SESSION_SECRET,
  resave: false,
  saveUninitialized: false,
  cookie: { httpOnly: true, sameSite: 'lax' }
}));
app.get('/pageview', (req, res) => {
  req.session.numviews = (req.session.numviews || 0) + 1;
  res.json({ views: req.session.numviews });
});

เพิ่ม SESSION_SECRET เป็นค่าสุ่มยาวใน .env ก่อนรัน สไลด์ใช้ saveUninitialized: true; ตัวอย่างนี้เลือก false เพื่อไม่บันทึก session ที่ยังไม่เปลี่ยนแปลง ส่วน resave: false ไม่บังคับบันทึกซ้ำเมื่อไม่มีการแก้ ทั้งนี้การต่ออายุขึ้นกับ store ด้วย MemoryStore เริ่มต้นเหมาะกับพัฒนา เมื่อ restart ข้อมูลในหน่วยความจำหาย จึงไม่ใช้เป็น session store ของระบบจริง

คาดการณ์ก่อนลอง: เปิดหน้าเดิมสองครั้งแล้วลบ cookie ตัวนับครั้งถัดไปจะต่อจากเดิมหรือเริ่มใหม่? แล้วถ้า cookie ยังอยู่แต่ store หายจะเกิดอะไรขึ้น?

ตามรอย ID และตัวนับ

Browser: ยังไม่มี ID
Server: {}

เมื่อส่งครั้งแรกได้ 1 ครั้งถัดไปได้ 2 ลบ cookie หรือ store แล้ว request ใหม่เริ่ม 1

แบบจำลองเพื่อเรียนรู้บนหน้านี้ ไม่ส่งข้อมูลจริง ไม่เชื่อม server หรือฐานข้อมูล และไม่บันทึกข้อมูลส่วนบุคคล

ลบ cookie ทำให้ browser หา session เดิมไม่เจอ แต่ record เดิมอาจยังอยู่จน store ล้างตามอายุ ส่วน logout ที่สมบูรณ์ต้องทำลาย session ฝั่ง server และล้าง cookie ที่เกี่ยวข้อง ไม่ใช่ลบตัวแปรในหน้าเว็บอย่างเดียว แบบจำลองนี้ย่อขั้นตอนการ save และสร้าง ID เพื่อให้เห็นความสัมพันธ์

L11 · หน้า PDF 35–44

ทำไมต้องมีฐานข้อมูลนอก session

Session เหมาะกับบริบทผู้ใช้ชั่วคราว แต่ตารางรายวิชาต้องเก็บอย่างเป็นระบบและมีความสัมพันธ์ Relational database เก็บข้อมูลเป็นตาราง เชื่อมกันด้วย keys และใช้ SQL เลือกหรือแก้ข้อมูล MySQL Server คือระบบที่รับ query; MySQL Workbench คือเครื่องมือให้คนจัดการ server ไม่ใช่ตัวฐานข้อมูลที่ Express ต้องเปิดหน้าต่างไว้

ตัวอย่าง tinycollege มี Department, Professor, Student, Course, Class และ Enroll: อาจารย์ทำงานในภาควิชาและเป็นหัวหน้าได้ นักศึกษาสังกัดภาคและมีอาจารย์ที่ปรึกษา Course เปิดโดยภาควิชา แต่ Class คือกลุ่มเรียนของ Course ที่มีอาจารย์สอน นักศึกษาหลายคนลงได้หลาย Class จึงใช้ Enroll เชื่อม ไม่ควรเก็บรายชื่อนักศึกษาทั้งหมดเป็นข้อความในแถว Class

เตรียม schema/tables ด้วย script หรือ Workbench แล้วสร้างบัญชีแอปที่ระบุ user/password/host ให้สิทธิ์เฉพาะ schema และ SQL ที่ต้องใช้ การใช้บัญชี root เพื่อจัดการครั้งแรกไม่ได้หมายความว่าควรให้เว็บใช้ root ตลอด

L11 · หน้า PDF 39–50

จาก request ผ่าน query กลับเป็น JSON

SELECT อ่านข้อมูล, INSERT เพิ่ม, UPDATE แก้, DELETE ลบ; WHERE กรองแถว, GROUP BY จัดกลุ่ม, HAVING กรองกลุ่ม และ ORDER BY เรียงผล ตัวอย่างนี้อ่านชื่อและรหัสวิชาเฉพาะภาค CIS

const mysql = require('mysql2');
const connection = mysql.createConnection({
  host: process.env.MYSQL_HOST,
  user: process.env.MYSQL_USERNAME,
  password: process.env.MYSQL_PASSWORD,
  database: process.env.MYSQL_DATABASE
});
app.get('/cis-courses', (req, res, next) => {
  connection.query(
    'SELECT course_code, course_name FROM Course WHERE dept_code = ?',
    ['CIS'],
    (err, rows) => {
      if (err) return next(err);
      res.json(rows);
    }
  );
});
app.use((err, req, res, next) => {
  res.status(500).json({ error: true, message: 'Database operation failed' });
});

callback ได้ error หรือผล query เมื่อฐานข้อมูลตอบกลับ จึงต้องส่ง response ภายใน callback หลังตรวจ error ไม่ใช่ส่ง rows ก่อนงานเสร็จ SELECT ได้อาร์เรย์แถว; INSERT ได้ผลการเขียน เช่น affectedRows และอาจมี insertId ไม่ใช่ชุดแถวแบบ SELECT

// ตัวอย่าง SQL สำหรับเพิ่มรายวิชา: ส่งค่าผ่าน placeholders
connection.query(
  'INSERT INTO Course (course_code, course_name, crs_credit, dept_code) VALUES (?, ?, ?, ?)',
  ['CIS-223', 'Introduction to Web Development', 3, 'CIS'],
  (err, result) => { /* ตรวจ err แล้วอ่าน result.affectedRows */ }
);

สไลด์หน้า 49 ใช้ GET เพิ่มข้อมูลเพื่อสาธิต และ redirect /cis-course ไม่ตรง route /cis-courses บทเรียนแยกให้เห็นว่าระบบจริงควรใช้ POST สำหรับเพิ่มข้อมูล และใช้ path ให้ตรงกัน อย่าต่อ input ผู้ใช้ลง SQL โดยตรง; mysql2 มี Promise API สำหรับ async/await ด้วย

เลือกที่เก็บให้ตรงอายุและหน้าที่

ข้อมูลที่เก็บที่เหมาะเพราะอะไร
สีธีมcookie หรือ localStorageให้ client จำความชอบ
จำนวนเข้าชมใน sessionsession storeผูกกับ session ID ของผู้ใช้
รายวิชาที่เปิดสอนฐานข้อมูลข้อมูลถาวรและสัมพันธ์กัน
รหัสผ่าน DB / secretenvironment configurationใช้ฝั่ง server ไม่ส่งเป็น JSON กลับไป

ลองอธิบายว่า “cookie ถูกลบ” ต่างจาก “ฐานข้อมูลถูกลบ” อย่างไร ถ้าคิดว่าตัวนับและรายวิชาจะหายพร้อมกัน ให้กลับไปแยก browser, session store และ MySQL อีกครั้ง
