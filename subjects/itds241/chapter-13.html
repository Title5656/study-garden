ลอง CRUD แล้วดูข้อมูลก่อน–หลัง

CRUD คือ Create/Read/Update/Delete ซึ่งสัมพันธ์กับ INSERT/SELECT/UPDATE/DELETE การตอบ success ต้องสะท้อนสิ่งที่เกิดขึ้น เช่น SELECT ไม่พบแถวควรอธิบายว่าไม่พบ ส่วน id ซ้ำเป็นคนละเหตุผลกับ body ไม่ครบ

ลอง GET id 1 → POST id 2 → เปลี่ยนชื่อด้วย PUT → DELETE id 2 แล้วลบซ้ำ คาดการณ์ทั้ง status และข้อมูลที่เหลือ แบบจำลองเลือกสถานะ 201/400/404/409 เพื่อให้แยกเหตุผลได้ สไลด์หลาย route ใช้ค่าเริ่มต้น 200 แม้เป็นการสร้าง

Student API ในหน่วยความจำ

แบบจำลองเพื่อเรียนรู้บนหน้านี้ ไม่ส่งข้อมูลจริง ไม่เชื่อม server หรือฐานข้อมูล และไม่บันทึกข้อมูลส่วนบุคคล

อ่านโค้ด service เป็นลำดับ ไม่ข้าม validation

ก่อนอ่าน req.body ต้องมี express.json() ก่อน routes; ก่อนส่ง SQL ต้องตรวจรูปข้อมูล แล้วส่ง input แยกด้วย placeholders ตัวอย่างต่อไปเป็น GET ที่ใช้ schema student ตามสไลด์ และใช้ Promise API ของ mysql2 เพื่อเห็นลำดับ await ชัดเจน

// server.cjs: ตั้ง MYSQL_* ใน .env และเตรียมตาราง student ก่อน
require('dotenv').config();
const express = require('express');
const mysql = require('mysql2/promise');
const app = express();
app.use(express.json());
const db = mysql.createPool({
  host: process.env.MYSQL_HOST,
  user: process.env.MYSQL_USERNAME,
  password: process.env.MYSQL_PASSWORD,
  database: process.env.MYSQL_DATABASE
});
app.get('/student/:id', async (req, res, next) => {
  if (!/^[1-9][0-9]*$/.test(req.params.id))
    return res.status(400).json({ error: true, message: 'Invalid id' });
  const id = Number(req.params.id);
  if (!Number.isSafeInteger(id))
    return res.status(400).json({ error: true, message: 'Invalid id' });
  try {
    const [rows] = await db.execute(
      'SELECT STU_ID, STU_FNAME, STU_LNAME, STU_AGE FROM student WHERE STU_ID = ?',
      [id]
    );
    if (!rows.length)
      return res.status(404).json({ error: true, message: 'Student not found' });
    res.json({ error: false, data: rows[0], message: 'Student retrieved' });
  } catch (err) { next(err); }
});
app.use((err, req, res, next) => {
  res.status(500).json({ error: true, message: 'Service operation failed' });
});
app.listen(Number(process.env.PORT || 3000));

POST ต้องตรวจ object และอนุญาตเฉพาะ STU_ID, STU_FNAME, STU_LNAME, STU_AGE แทนรับทุก key ไปใส่ SQL; UPDATE ต้องระบุ WHERE ให้ตรง record; DELETE ต้องตรวจ affectedRows; อย่า throw ใน callback แล้วคาดว่า Express จะจับให้เอง ให้ next(err) ไป error middleware

// รูป SQL สำหรับเขียน: ใช้ค่าที่ผ่าน validation แล้วเท่านั้น
INSERT INTO student (STU_ID, STU_FNAME, STU_LNAME, STU_AGE) VALUES (?, ?, ?, ?)
UPDATE student SET STU_FNAME = ?, STU_LNAME = ?, STU_AGE = ? WHERE STU_ID = ?
DELETE FROM student WHERE STU_ID = ?

ทดสอบให้รู้ว่าเสียที่สัญญาหรือ client

ใน Postman สร้าง request เลือก method ใส่ URL ให้ตรง ถ้าเป็น JSON เลือก body แบบ raw/JSON และ Content-Type: application/json แล้วส่ง ตรวจ status, headers และ body รวมถึงตรวจฐานข้อมูลหลังคำขอเขียน อย่าทดสอบเพียง 200 หนึ่งครั้ง

กรณีสิ่งที่ควรตรวจ
GET ID ที่มี / ไม่มีได้ record ถูกคน / ไม่พบตามสัญญา
POST ข้อมูลครบ / ขาด fieldเพิ่มได้ / ได้ error โดยไม่เพิ่มแถว

ฝั่งฟอร์มอ่านค่าจาก input แล้วเรียก service แบบ asynchronous เมื่อรับ data แล้วจึงใส่ค่ากลับฟอร์ม ต้องตรวจ error/not-found ก่อนเข้าถึง data.STU_ID ไม่เช่นนั้นจะเกิดข้อผิดพลาดซ้ำที่ UI

async function callStudentWS(url, method, sentData) {
  const options = { method, headers: { Accept: 'application/json' } };
  if (sentData !== undefined && method !== 'GET') {
    options.headers['Content-Type'] = 'application/json';
    options.body = JSON.stringify(sentData);
  }
  const response = await fetch(url, options);
  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  const result = await response.json();
  if (result.error) throw new Error(result.message);
  return result;
}
// ตัวอย่าง: จับข้อผิดพลาดก่อนอัปเดต UI
callStudentWS('/student/4', 'GET').then(result => {
  document.querySelector('#student-name').value = result.data.STU_FNAME;
}).catch(err => { console.error(err.message); });

CORS: ใครอนุญาตให้ browser อ่าน response

Origin ประกอบด้วย scheme, host และ port เช่น http://localhost:3100 กับ http://localhost:3000 ต่าง origin แม้ host เหมือนกัน Same-origin policy จำกัดการอ่านข้อมูลข้าม origin ของ scripts; CORS ให้ server ประกาศสิ่งที่อนุญาตผ่าน headers ไม่ใช่สิทธิ์ที่ client เปิดเองได้

const cors = require('cors');
app.use(cors({
  origin: 'http://localhost:3100',
  methods: ['GET', 'POST', 'PUT', 'DELETE']
})); // วางก่อน routes
// หรือกำหนดเฉพาะ route: app.get('/student/:id', cors(options), handler)

app.use(cors()) ตาม demo เปิดอ่านแบบทั่วไปกว้างกว่า allowlist คำขอบางประเภท เช่น PUT, JSON POST หรือ Authorization header ต้องมี preflight OPTIONS ก่อนจริง ถ้า preflight ไม่ผ่าน browser ไม่ส่งคำขอจริง แต่ simple request อาจถึง server ก่อนถูกบล็อกการอ่าน response Postman/Node ไม่อยู่ภายใต้ CORS แบบ browser จึงอาจเรียกได้แม้หน้าเว็บอ่านไม่ได้

พอร์ตเดียวกันไหม แล้วอ่านได้หรือไม่?

หน้าเว็บพอร์ต 3100 เรียก API พอร์ต 3000 ต้องได้รับ CORS headers ที่ถูกต้อง

แบบจำลองเพื่อเรียนรู้บนหน้านี้ ไม่ส่งข้อมูลจริง ไม่เชื่อม server หรือฐานข้อมูล และไม่บันทึกข้อมูลส่วนบุคคล

เมื่อส่ง credentials ข้าม origin จะใช้ Allow-Origin: * ไม่ได้ ต้องระบุ origin และ allow-credentials ให้ถูก แบบจำลองตรวจเฉพาะ origin/credentials ของ response และสมมติ preflight ผ่านเมื่อจำเป็น ไม่ได้ทดสอบเครือข่ายจริง CORS ไม่แทน authentication, authorization หรือ CSRF protection

JWT: อ่าน payload ได้ ไม่ได้แปลว่าเชื่อถือได้

Authentication ตรวจว่าเป็นใคร ส่วน authorization ตรวจว่าทำอะไรได้ ใน token-based flow ผู้ใช้ login → server ตรวจ credentials → ออก token → client ส่ง Authorization: Bearer token ทุกครั้ง → middleware ตรวจ → จึงเข้า handler

JWT แบบ signed token ที่สไลด์สอนมีสามส่วนคั่นด้วยจุด header.payload.signature; header/payload เป็น Base64URL จึงอ่านได้ ลายเซ็นช่วยตรวจความถูกต้องและการแก้ไข ไม่ใช่การปกปิดข้อมูล Claims มี registered เช่น iss, sub, aud, exp, iat; public ที่ตั้งแบบไม่ชนกัน และ private ที่คู่สื่อสารตกลงร่วมกัน ห้ามใส่รหัสผ่านลง payload

const jwt = require('jsonwebtoken');
// ทำหลังตรวจ credentials สำเร็จ โดยใช้ user จากแหล่งที่เชื่อถือได้
const token = jwt.sign({ sub: String(user.id) }, process.env.JWT_SECRET, {
  algorithm: 'HS256', expiresIn: '1h', issuer: 'student-api', audience: 'student-ui'
});
function authorize(req, res, next) {
  const header = req.headers.authorization || '';
  const match = /^Bearer ([^ ]+)$/.exec(header);
  if (!match) return res.status(401).json({ error: true, message: 'Token required' });
  try {
    req.auth = jwt.verify(match[1], process.env.JWT_SECRET, {
      algorithms: ['HS256'], issuer: 'student-api', audience: 'student-ui'
    });
    next();
  } catch {
    res.status(401).json({ error: true, message: 'Invalid or expired token' });
  }
}
// app.get('/students', authorize, handler);

โค้ดนี้เป็นส่วนประกอบ ไม่ใช่ระบบ login ครบชุด ต้องกำหนด user จากการตรวจ credentials ก่อน sign และตั้ง JWT_SECRET ฝั่ง server สไลด์ /signin รับ email/userid จาก body แล้ว sign เพื่อสาธิต จึงยังไม่พิสูจน์ตัวตน; อย่าใช้รูปนั้นเป็น login จริง หน้า PDF 65 ยังสลับลำดับ method/token ของ callStudentWS ให้รักษาลำดับอาร์กิวเมนต์ให้ตรงกับนิยาม

แยก token ใช้ไม่ได้ กับไม่มีสิทธิ์

ไม่ส่ง token ได้ 401; token ผ่านแต่ไม่มีสิทธิ์ได้ 403

แบบจำลองเพื่อเรียนรู้บนหน้านี้ ไม่ส่งข้อมูลจริง ไม่เชื่อม server หรือฐานข้อมูล และไม่บันทึกข้อมูลส่วนบุคคล

ฝั่ง client ส่ง headers: { Authorization: 'Bearer ' + token } การเลือกที่เก็บ token มีข้อแลกเปลี่ยน: local/sessionStorage อ่านได้ด้วย JavaScript จึงเสี่ยงถูกขโมยเมื่อมี XSS; HttpOnly cookie จำกัดการอ่านด้วย JS แต่ต้องออกแบบ SameSite/CSRF และ credentials; เก็บใน memory ลดการคงค้างแต่ refresh แล้วหาย JWT ไม่ได้แก้ปัญหาเหล่านี้เอง และ logout ที่ client ลบ token ไม่ทำให้ token ที่ถูกคัดลอกหมดอายุทันที

Node.js เป็นผู้เรียกบริการอื่นได้ด้วย Axios

บริการหนึ่งอาจเรียกอีกบริการ เช่นค้นสูตรอาหารหรือเรียก Student WS Axios เป็น HTTP client สำหรับ browser/Node ใช้ get/delete โดย config เป็นอาร์กิวเมนต์ที่สอง ส่วน post/put/patch รับ data ก่อน config เมื่อได้ response ตัวข้อมูลอยู่ใน response.data ต่างจาก fetch ที่ต้องเรียก response.json()

const axios = require('axios');
async function getStudents() {
  try {
    const response = await axios.get('http://localhost:3000/students', {
      headers: { Authorization: `Bearer ${process.env.ACCESS_TOKEN}` },
      timeout: 5000
    });
    console.log(response.data);
  } catch (err) {
    console.error(err.response ? `HTTP ${err.response.status}` : 'Network/timeout error');
  }
}
getStudents();

ตัวอย่าง spoonacular ส่ง query, number และ apiKey ผ่าน config.params แทนต่อ URL เอง; เก็บ API key ใน environment ฝั่ง server ไม่ใส่ source ที่ส่งให้ browser ความพร้อมใช้และข้อจำกัดของบริการภายนอกเป็นอีกเรื่อง ตัวบทไม่เรียก API จริง

const response = await axios.get(
  'https://api.spoonacular.com/recipes/complexSearch',
  { params: { query: 'pizza', number: 20, apiKey: process.env.SPOONACULAR_API_KEY }, timeout: 5000 }
);
console.table(response.data.results);

Axios มีการแปลง JSON, timeout, cancellation และกลไกช่วย XSRF ตาม configuration แต่ไม่ใช่การป้องกัน CSRF อัตโนมัติทั้งระบบ ต้องออกแบบฝั่ง server ด้วย ส่วนแพ็กเกจ HTTP clients อื่นในสไลด์เป็นรายการตัวอย่าง ไม่ใช่ข้อกำหนดว่าต้องติดตั้งทั้งหมด

Postman ได้ แต่ browser อ่านไม่ได้origin, preflight, CORS headers
401Bearer header, signature, expiry, issuer/audience
403สิทธิ์ของผู้ใช้ที่ตรวจตัวตนผ่านแล้ว

สิ่งที่ควรอธิบายได้หลังบทนี้คือ request หนึ่งผ่าน parser → CORS/การตรวจสิทธิ์ตามลำดับ middleware → validation → query → response อย่างไร และแต่ละจุดผิดแล้วควรส่งผลแบบใด คำตอบต้องเชื่อมกับสัญญา API ไม่ใช่ท่องเพียงชื่อ package

เอกสารทางการสำหรับตรวจพฤติกรรมและอ่านต่อ: MDN CORS · jsonwebtoken · Axios
